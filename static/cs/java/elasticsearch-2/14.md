### DSL聚合
#### 1. Bucket聚合
* 样例： 统计所有商品中共有哪些商品分类
```
    GET /demo/_search
    {
      "query": {"match_all": {}},    // 可以省略
      "size": 0,                     // 设置size为0，结果中不包含文档，只包含聚合结果
      "aggs": {                      // 定义聚合
        "category_agg": {            // 给聚合起个名字
          "terms": {                 // 聚合的类型，按照商品分类（category）聚合，所以选择term
            "field": "category",     // 参与聚合的字段
            "size": 20               // 希望获取的聚合结果数量
          }
        },
        "brand_agg": {
          "terms": {
            "field": "brand",
            "size": 20
          }
        }
      }
    }
```

#### 2. 带条件的聚合
```
    GET /demo/_search
    {
      "query": {
        "bool": {
          "filter": [
            {
              "term": {
                "category": "手机"
              }
            },
            {
              "range": {
                "price": {
                  "gte": 300000
                }
              }
            }
          ]
        }
      },
      "size": 0,
      "aggs": {
        "brand_agg": {
          "terms": {
            "field": "brand",
            "size": 20
          }
        }
      }
    }
```

#### 3. Metric聚合
```
    GET /demo/_search
    {
      "query": {
        "bool": {
          "filter": [
            {
              "term": {
                "category": "手机"
              }
            },
            {
              "range": {
                "price": {
                  "gte": 300000
                }
              }
            }
          ]
        }
      },
      "size": 0,
      "aggs": {
        "brand_agg": {
          "terms": {
            "field": "brand",
            "size": 20
          },
          "aggs": {
            "stats_meric": {
              "stats": {
                "field": "price"
              }
            }
          }
        }
      }
    }
```
