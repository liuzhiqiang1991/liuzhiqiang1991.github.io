```
    <bean id="demo-ds" class="com.zaxxer.hikari.HikariDataSource" destroy-method="close">

        <property name="jdbcUrl" value="jdbc:mysql://localhost:3306/sharding-jdbc"/>
        <property name="username" value="root"/>
        <property name="password" value="root"/>
    </bean>
```
