### 分库
#### 1. pom
```
    <!--分库分表-->
    <dependency>
        <groupId>io.shardingsphere</groupId>
        <artifactId>sharding-jdbc-core</artifactId>
        <version>3.0.0</version>
    </dependency>
    <dependency>
        <groupId>io.shardingjdbc</groupId>
        <artifactId>sharding-jdbc-core-spring-namespace</artifactId>
        <version>2.0.3</version>
    </dependency>
```

#### 2. log-datasource.xml
```
    xmlns:sharding="http://shardingjdbc.io/schema/shardingjdbc/sharding"
    http://shardingjdbc.io/schema/shardingjdbc/sharding
    http://shardingjdbc.io/schema/shardingjdbc/sharding/sharding.xsd


    <context:property-placeholder ignore-unresolvable="true"/>
    <!--数据库sharding策略-->
    <sharding:standard-strategy id="logStrategy" sharding-column="product_id"
                                precise-algorithm-class="com.demo.sharding.LogShardingAlgorithm"/>
    <!--sharding策略:不分片的策略 -->
    <sharding:none-strategy id="noStrategy"/>

    <sharding:data-source id="logShardingDataSource">
        <sharding:sharding-rule
                data-source-names="demo-log-ds0,demo-log-ds1"
                default-database-strategy-ref="noStrategy"
                default-table-strategy-ref="noStrategy">
            <sharding:table-rules>
                <!--只分库不分表-->
                <sharding:table-rule logic-table="log"
                                     actual-data-nodes="demo-log-ds${0..1}.log"
                                     database-strategy-ref="logStrategy"/>
            </sharding:table-rules>
        </sharding:sharding-rule>
        <sharding:props>
            <prop key="sql.show">true</prop>
            <prop key="executor.size">100</prop>
        </sharding:props>
    </sharding:data-source>


    <bean id="demoLogSqlSessionFactory" class="org.mybatis.spring.SqlSessionFactoryBean">
        <property name="configLocation" value="classpath:conf/mybatis-config-log.xml"/>
        <property name="dataSource" ref="logShardingDataSource"/>
    </bean>

    <bean id="demoLogSqlSession" class="org.mybatis.spring.SqlSessionTemplate">
        <constructor-arg index="0" ref="demoLogSqlSessionFactory"/>
    </bean>

    <bean id="demoLogDataSourceManager" class="org.springframework.jdbc.datasource.DataSourceTransactionManager">
        <property name="dataSource" ref="logShardingDataSource"/>
    </bean>
```

#### 3. LogShardingAlgorithm
```

    import io.shardingjdbc.core.api.algorithm.sharding.PreciseShardingValue;
    import io.shardingjdbc.core.api.algorithm.sharding.standard.PreciseShardingAlgorithm;
    import lombok.extern.slf4j.Slf4j;
    import java.util.Collection;

    @Slf4j
    public class LogShardingAlgorithm implements PreciseShardingAlgorithm<Long> {

        private static final int SHARDING_NUMBER = 2;

        @Override
        public String doSharding(Collection<String> collection, PreciseShardingValue<Long> preciseShardingValue) {
            for (String each : collection) {
                if (each.endsWith(preciseShardingValue.getValue() % SHARDING_NUMBER + "")) {
                    log.info("the target table name:"+each);
                    return each;
                }
            }
            throw new IllegalArgumentException();
        }
    }
```
