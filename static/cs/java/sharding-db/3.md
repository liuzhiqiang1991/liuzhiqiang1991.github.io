#### 3. conf/mapper/log/LogMapper.xml
```
    <!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "http://mybatis.org/dtd/mybatis-3-mapper.dtd">

    <mapper namespace="com.demo.dao.LogDao">

        <insert id="insert" parameterType="com.demo.po.Log">
            insert into log (product_id, content)
            values (#{productId, jdbcType=BIGINT}, #{content, jdbcType=VARCHAR} )
        </insert>

        <select resultType="com.demo.po.Log" id="list">
            select * from log
        </select>
    </mapper>
```

#### 5. dao
```
    import com.demo.po.Log;
    import java.util.List;

    public interface LogDao {

        int insert(Log log);

        List<Log> list();

    }




    import com.demo.dao.LogDao;
    import com.demo.po.Log;
    import org.apache.ibatis.session.SqlSession;
    import org.springframework.stereotype.Repository;
    import javax.annotation.Resource;
    import java.util.List;

    @Repository
    public class LogDaoImpl implements LogDao {

        @Resource(name = "demoLogSqlSession")
        private SqlSession sqlSession;

        private String namespace = LogDao.class.getName();

        @Override
        public int insert(Log user) {
            return sqlSession.insert(namespace + ".insert", user);
        }

        @Override
        public List<Log> list() {
            return sqlSession.selectList(namespace + ".list");
        }
    }
```

#### 6. controller
```
    import com.demo.dao.LogDao;
    import com.demo.po.Log;
    import org.springframework.web.bind.annotation.GetMapping;
    import org.springframework.web.bind.annotation.PostMapping;
    import org.springframework.web.bind.annotation.RequestBody;
    import org.springframework.web.bind.annotation.RestController;
    import javax.annotation.Resource;
    import java.util.List;

    @RestController
    public class LogController {

        @Resource
        private LogDao logDao;

        @GetMapping("/log/list")
        public List<Log> list() {
            return logDao.list();
        }

        @PostMapping("/log/add")
        public void add(@RequestBody Log log) {
            logDao.insert(log);
        }
    }
```

    @Target(ElementType.TYPE)
    @Retention(RetentionPolicy.RUNTIME)

    }
