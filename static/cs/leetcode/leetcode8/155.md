#### 155. 最小栈
```
    class MinStack {

        private Stack<Integer> data = new Stack<>();
        private Stack<Integer> min = new Stack<>();

        public MinStack() {

        }

        public void push(int val) {
            if(data.size() == 0) {
                data.push(val);
                min.push(val);
            } else {
                data.push(val);
                int t = min.peek();
                min.push(Math.min(t, val));
            }
        }

        public void pop() {
            data.pop();
            min.pop();
        }

        public int top() {
            return data.size() > 0 ? data.peek() : 0;
        }

        public int getMin() {
            return min.size() > 0 ? min.peek() : 0;
        }
    }
```
