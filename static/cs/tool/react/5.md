#### Multiple Components
#### 1. Navbar.js
```
    const Navbar = () => {
        return (
            <nav className="navbar">
                <h1>The Blog</h1>
                <div className="links">
                    <a href="/">Home</a>
                    <a href="create">New Blog</a>
                </div>
            </nav>
        );
    }
    export default Navbar;
```

#### 2. Home.js
```
    const Home = () => {
        return (
            <div className="home">
                <h2>Homepage</h2>
            </div>
        );
    }
    export default Home;
```

#### 3. App.js
```
    import './App.css';
    import Navbar from "./Navbar";
    import Home from './Home';

    function App() {
      return (
        <div className="App">
            <Navbar />
            <div className="content">
                <Home />
            </div>
        </div>
      );
    }

    export default App;
```
