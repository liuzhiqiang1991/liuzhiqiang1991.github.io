#### Dynamic Values in Templates
#### 1. App.js
```
    import './App.css';

    function App() {
      const title = "测试";
      const number = 10;

      return (
        <div className="App">
          <div className="content">
            <h1>Component Test</h1>
            <h2>{ title } : { Math.random() * number }</h2>
          </div>
        </div>
      );
    }

    export default App;
```
