#### useEffect Hook
#### 1. Home.js
```
    import { useState, useEffect } from 'react';
    import BlogList from './BlogList';

    const Home = () => {

        const [blogs, setBlogs] = useState([
            { id: 1, title: 'My First Blog', author: 'tom', body: 'Hello, everyone'},
            { id: 2, title: 'The Beautiful', author: 'Jim', body: 'Beautiful color'},
            { id: 3, title: 'Crash courses', author: 'Ben', body: 'How to learn...'}
        ]);

        const handleDelete = (id) => {
            const tmp = blogs.filter(blog => blog.id !== id);
            setBlogs(tmp);
        }

        useEffect(() => {
            console.log('use effect')
        });

        return (
            <div className="home">
                <BlogList blogs={blogs} title="All Blogs" handleDelete={handleDelete} />
            </div>
        );
    }
    export default Home;
```
