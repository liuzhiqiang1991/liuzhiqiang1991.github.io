#### Outputting Lists
#### 1. Home.js
```
    import { useState } from 'react';

    const Home = () => {

        const [blogs, SetBlogs] = useState([
            { id: 1, title: 'My First Blog', author: 'tom', body: 'Hello, everyone'},
            { id: 2, title: 'The Beautiful', author: 'Jim', body: 'Beautiful color'},
            { id: 3, title: 'Crash courses', author: 'Ben', body: 'How to learn...'}
        ]);

        return (
            <div className="home">
                {blogs.map((blog) => (
                    <div className="blog-preview" key={blog.id}>
                        <h2>{ blog.title }</h2>
                        <p>Written by { blog.author }</p>
                    </div>
                ))}
            </div>
        );
    }
    export default Home;
```
