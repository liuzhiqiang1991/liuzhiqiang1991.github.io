#### Functions as Props
* pass functions to a component as props
#### 1. Home.js
```
    import { useState } from 'react';
    import BlogList from './BlogList';

    const Home = () => {

        const [blogs, setBlogs] = useState([
            { id: 1, title: 'My First Blog', author: 'tom', body: 'Hello, everyone'},
            { id: 2, title: 'The Beautiful', author: 'Jim', body: 'Beautiful color'},
            { id: 3, title: 'Crash courses', author: 'Ben', body: 'How to learn...'}
        ]);

        const handleDelete = (id) => {
            const tmp = blogs.filter(blog => blog.id !== id);
            setBlogs(tmp);
        }

        return (
            <div className="home">
                <BlogList blogs={blogs} title="All Blogs" handleDelete={handleDelete} />
            </div>
        );
    }
    export default Home;
```

#### 2. BlogList.js
```
    const BlogList = (props) => {
        const blogs = props.blogs;
        const title = props.title;
        const handleDelete = props.handleDelete;

        return (
            <div className="blog-list">
                <h2>{ title }</h2>
                {blogs.map((blog) => (
                    <div className="blog-preview" key={blog.id}>
                        <h2>{ blog.title }</h2>
                        <p>Written by { blog.author }</p>
                        <button onClick={() => handleDelete(blog.id)}>delete blog</button>
                    </div>
                ))}
            </div>
        );
    }

    export default BlogList;
```
