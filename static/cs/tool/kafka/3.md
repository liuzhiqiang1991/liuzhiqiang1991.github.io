### Kafka Producers

#### 1. 项目初始化
#### 2. pom
```
        <!-- Spring Kafka -->
        <dependency>
            <groupId>org.springframework.kafka</groupId>
            <artifactId>spring-kafka</artifactId>
        </dependency>
```

#### 3. application.yaml
```
    spring:
      kafka:
        bootstrap-servers: localhost:9092
        producer:
          key-serializer: org.apache.kafka.common.serialization.StringSerializer
          value-serializer: org.apache.kafka.common.serialization.StringSerializer
```

#### 4. 发送消息
```
    import org.junit.jupiter.api.Test;
    import org.springframework.beans.factory.annotation.Autowired;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.kafka.core.KafkaTemplate;

    @SpringBootTest
    public class KafkaProducer {

        @Autowired
        private KafkaTemplate kafkaTemplate;

        @Test
        public void test() {
            kafkaTemplate.send("demo-tom", "hello, tom");
        }

        @Test
        public void testSynchronize() throws Exception {
            ListenableFuture<SendResult<String, String>> future =
                    kafkaTemplate.send("demo-tom", "test synchronize");
            SendResult<String, String> res = future.get();
            ProducerRecord<String, String> record = res.getProducerRecord();
            log.info("record: {}", record);
        }

        @Test
        public void testASynchronize() throws Exception {
            ListenableFuture<SendResult<String, String>> future =
                    kafkaTemplate.send("demo-tom", "test ASynchronize 007");
            future.addCallback(new ListenableFutureCallback<SendResult<String, String>>() {
                @Override
                public void onFailure(Throwable ex) {
                    log.info("处理失败");
                }

                @Override
                public void onSuccess(SendResult<String, String> result) {
                    log.info("处理成功，结果：{}", result);
                    log.info("recordMetadata: {}", result.getRecordMetadata());
                    RecordMetadata metadata = result.getRecordMetadata();
                    log.info("topic: {}, partition: {}, offset: {}", metadata.topic(), metadata.partition(), metadata.offset());
                }
            });
            Thread.sleep(1000);
        }
    }
```
