#### 404 pages
#### 1. NotFound.js
```
    import { Link } from "react-router-dom"

    const NotFound = () => {
        return (
            <div className="not-found">
                <h2>Sorry</h2>
                <p>That page cannot be found</p>
                <Link to="/">Back to the homepage...</Link>
            </div>
        )
    }

    export default NotFound;
```

#### 2. App.js
```
    import NotFound from './NotFound';

    <Route path="*">
        <NotFound />
    </Route>
```
