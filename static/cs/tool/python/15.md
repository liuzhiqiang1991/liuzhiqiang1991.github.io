### 继承

<pre><code style="font-size: 18px; font-family: Microsoft YaHei; line-height: 31px;">
    class Mammal:
        def __init__(self, name, gender):
            self.name = name
            self.gender = gender
        def breath(self):
            print(self.name + "在呼吸...")
        def walk(self):
            print(self.name + "在走路...")

    class Human(Mammal):
        def __init__(self, name, gender):
            super().__init__(name, gender)
            self.has_tail = False
        def read(self):
            print(self.name + "在阅读...")

    class Cat(Mammal):
        def __init__(self, name, gender):
            super().__init__(name, gender)
            self.has_tail = True
        def walk(self):
            print(self.name + " 小猫在走路...")

    cat1 = Cat("Tom", "male")
    cat1.walk()
    man1 = Human("Green", "male")
    man1.read()
    man1.walk()
</code></pre>
