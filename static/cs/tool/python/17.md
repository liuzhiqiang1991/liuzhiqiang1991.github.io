#### 处理异常
<pre><code style="font-size: 18px; font-family: Microsoft YaHei; line-height: 31px;">
    try:
        user_weight = float(input("请输入体重（单位：kg）:"))
        user_height = float(input("请输入身高（单位：m）:"))
        user_BMI = user_weight / user_height ** 2
    except ValueError:
        print("输入不为合理数字，请重新运行程序")
    except ZeroDivisionError:
        print("身高不能为零，请重新运行程序")
    except:
        print("发生了未知错误，请重新运行程序")
    else:
        print("BMI值为：" + str(user_BMI))     # 没有错误时会运行
    finally:
        print("程序结束运行")     # 不管发生错误与否都会运行
</code></pre>
